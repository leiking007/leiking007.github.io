{{/* 相对路径拼接页面路径；以 / 开头的站内绝对路径与外链(http/https)原样使用 */}} {{- $imgUrl := .Destination -}} {{- $isRelative := and (not (strings.HasPrefix $imgUrl "/")) (not (urls.Parse $imgUrl).IsAbs) -}} {{- if $isRelative -}} {{- $imgUrl = printf "%s%s" .Page.RelPermalink $imgUrl -}} {{- end -}} {{/* 页面资源图片输出实际宽高，避免加载时布局偏移(CLS) */}} {{- $res := "" -}} {{- if and $isRelative (not (strings.Contains .Destination "?")) -}} {{- $res = .Page.Resources.GetMatch (strings.TrimPrefix "./" .Destination) -}} {{- end -}}
