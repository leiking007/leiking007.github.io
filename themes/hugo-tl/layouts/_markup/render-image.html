{{/* 相对路径拼接页面路径；以 / 开头的站内绝对路径与外链(http/https)原样使用 */}} {{- $imgUrl := .Destination -}} {{- $isRelative := and (not (strings.HasPrefix $imgUrl "/")) (not (urls.Parse $imgUrl).IsAbs) -}} {{- if $isRelative -}} {{- $imgUrl = printf "%s%s" .Page.RelPermalink $imgUrl -}} {{- end -}} {{/* 页面资源图片：转为 webp 并生成响应式 srcset，同时输出实际宽高避免布局偏移(CLS) */}} {{- $res := "" -}} {{- if and $isRelative (not (strings.Contains .Destination "?")) -}} {{- $res = .Page.Resources.GetMatch (strings.TrimPrefix "./" .Destination) -}} {{- end -}} {{- $width := "" -}} {{- $height := "" -}} {{- $srcset := "" -}} {{- with $res -}} {{- $res = . -}} {{- if in (slice "jpeg" "png" "webp") .MediaType.SubType -}} {{- $fallback := .Process "webp q80" -}} {{- if gt .Width 1200 -}} {{- $fallback = .Resize "1200x q80 webp" -}} {{- end -}} {{- $imgUrl = $fallback.RelPermalink -}} {{- $width = $fallback.Width -}} {{- $height = $fallback.Height -}} {{- $variants := slice -}} {{- range $w := slice 480 768 1200 -}} {{- if ge $res.Width $w -}} {{- $variant := $res.Resize (printf "%dx q80 webp" $w) -}} {{- $variants = $variants | append (printf "%s %dw" $variant.RelPermalink $w) -}} {{- end -}} {{- end -}} {{- if ge (len $variants) 2 -}} {{- $srcset = delimit $variants ", " -}} {{- end -}} {{- else if gt .Width 0 -}} {{- $width = .Width -}} {{- $height = .Height -}} {{- end -}} {{- end -}}
